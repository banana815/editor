cmake_minimum_required(VERSION 3.24)
project(NoteLabIOS LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  message(FATAL_ERROR "Use build.sh, or configure with -DCMAKE_SYSTEM_NAME=iOS")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(EDITOR_USE_SYSTEM_SDL OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. notelab)
set_target_properties(notelab PROPERTIES
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
  MACOSX_BUNDLE_BUNDLE_NAME "NoteLab"
  MACOSX_BUNDLE_BUNDLE_VERSION "4"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.1"
  XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "org.notelab.editor"
  XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
)
