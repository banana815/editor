cmake_minimum_required(VERSION 3.24)
option(EDITOR_DISTRIBUTION "Build a self-contained desktop application archive" OFF)
if(EDITOR_DISTRIBUTION)
  # Apply consistently before creating dependency targets, including their C++
  # translation units. Windows packages need no separate VC runtime installer.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(EDITOR_USE_SYSTEM_SDL OFF CACHE BOOL "" FORCE)
endif()
project(NoteLab VERSION 0.2.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
  # Emscripten's JavaScript bridge macros require GNU C syntax in platform glue.
  set(CMAKE_C_EXTENSIONS ON)
else()
  set(CMAKE_C_EXTENSIONS OFF)
endif()

if(MSVC)
  add_compile_options(/utf-8)
endif()
option(EDITOR_BUILD_APP "Build the SDL desktop/mobile/browser application" ON)
option(EDITOR_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
include(CTest)
include(cmake/Dependencies.cmake)
include(cmake/EmojiFont.cmake)
include(cmake/SQLite.cmake)
include(cmake/Notices.cmake)

add_library(editor_core STATIC src/document.c src/note_store.c src/zotero.c src/citations.c src/library_panel.c)
set_target_properties(editor_core PROPERTIES C_EXTENSIONS OFF)
target_include_directories(editor_core PUBLIC include)
target_link_libraries(editor_core PUBLIC cjson editor_sqlite)
if(MSVC)
  target_compile_options(editor_core PRIVATE /W4)
else()
  target_compile_options(editor_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EDITOR_SANITIZERS AND NOT MSVC)
  target_compile_options(editor_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(editor_core PUBLIC -fsanitize=address,undefined)
endif()

if(BUILD_TESTING AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(test_document tests/test_document.c)
  target_link_libraries(test_document PRIVATE editor_core)
  add_test(NAME document COMMAND test_document)
  add_executable(test_note_store tests/test_note_store.c)
  target_link_libraries(test_note_store PRIVATE editor_core)
  add_test(NAME note_store COMMAND test_note_store)
  add_executable(test_zotero tests/test_zotero.c)
  target_link_libraries(test_zotero PRIVATE editor_core)
  if(MINGW)
    target_link_options(test_zotero PRIVATE -municode)
  endif()
  add_test(NAME zotero COMMAND test_zotero "${CMAKE_CURRENT_BINARY_DIR}")
  add_executable(test_citations tests/test_citations.c)
  target_link_libraries(test_citations PRIVATE editor_core)
  add_test(NAME citations COMMAND test_citations)
  add_executable(test_library_panel tests/test_library_panel.c)
  target_link_libraries(test_library_panel PRIVATE editor_core)
  add_test(NAME library_panel COMMAND test_library_panel)
  add_executable(test_persistence_native tests/test_platform_persistence.c src/platform_persistence.c)
  target_include_directories(test_persistence_native PRIVATE include)
  add_test(NAME persistence_native COMMAND test_persistence_native)
  if(NOT EMSCRIPTEN)
    add_executable(test_persistence_queue tests/test_platform_persistence.c src/platform_persistence.c)
    target_include_directories(test_persistence_queue PRIVATE include tests/support/emscripten_stub)
    target_compile_definitions(test_persistence_queue PRIVATE __EMSCRIPTEN__)
    add_test(NAME persistence_queue COMMAND test_persistence_queue)
    find_program(EDITOR_NODE_EXECUTABLE node)
    if(EDITOR_NODE_EXECUTABLE)
      add_test(NAME persistence_bridge COMMAND "${EDITOR_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_persistence_bridge.mjs")
      add_test(NAME import_bridge COMMAND "${EDITOR_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_import_bridge.mjs")
    endif()
  endif()
  foreach(test_target test_document test_note_store test_zotero test_citations test_library_panel test_persistence_native test_persistence_queue)
    if(NOT TARGET ${test_target})
      continue()
    endif()
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
      if(EDITOR_SANITIZERS)
        target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${test_target} PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
  endforeach()
endif()

if(EDITOR_BUILD_APP)
  configure_file("${EDITOR_FONT_PATH}" "${EDITOR_ASSETS_DIR}/font.otf" COPYONLY)
  if(ANDROID)
    add_library(notelab SHARED src/app.c src/platform_import.c src/platform_persistence.c src/platform_web_input.c src/platform_export.c src/rich_layout.c vendor/editbox.c)
    set_target_properties(notelab PROPERTIES OUTPUT_NAME main)
  else()
    add_executable(notelab src/app.c src/platform_import.c src/platform_persistence.c src/platform_web_input.c src/platform_export.c src/rich_layout.c vendor/editbox.c)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJC)
    target_sources(notelab PRIVATE src/platform_import_ios.m src/platform_export_ios.m)
    set_source_files_properties(src/platform_import_ios.m src/platform_export_ios.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(notelab PRIVATE "-framework UIKit" "-framework Foundation" "-framework UniformTypeIdentifiers")
  endif()
  target_include_directories(notelab PRIVATE vendor)
  target_link_libraries(notelab PRIVATE editor_core SDL3::SDL3 SDL3_ttf::SDL3_ttf)
  target_compile_definitions(notelab PRIVATE EDITOR_FONT_PATH="assets/font.otf")
  set_property(TARGET notelab APPEND PROPERTY LINK_DEPENDS
    "${EDITOR_FONT_PATH}" "${EDITOR_EMOJI_FONT_PATH}" "${EDITOR_EMOJI_LICENSE_PATH}"
    "${EDITOR_THIRD_PARTY_PATH}" ${EDITOR_NOTICE_FILES})
  if(NOT MSVC)
    target_compile_options(notelab PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING AND NOT ANDROID)
    if(APPLE)
      enable_language(OBJC)
      add_executable(test_import_types platform/ios/test_import_types.m)
      target_include_directories(test_import_types PRIVATE include)
      target_link_libraries(test_import_types PRIVATE SDL3::SDL3 "-framework Foundation" "-framework UniformTypeIdentifiers")
      target_compile_options(test_import_types PRIVATE -fobjc-arc -UNDEBUG -Wall -Wextra -Werror)
      add_test(NAME apple_import_types COMMAND test_import_types)
    endif()
    add_executable(test_platform_import tests/test_platform_import.c)
    target_include_directories(test_platform_import PRIVATE include)
    target_link_libraries(test_platform_import PRIVATE SDL3::SDL3)
    if(MSVC)
      target_compile_options(test_platform_import PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_platform_import PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
      if(EDITOR_SANITIZERS)
        target_compile_options(test_platform_import PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test_platform_import PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
    add_test(NAME platform_import COMMAND test_platform_import "${CMAKE_CURRENT_BINARY_DIR}")
    add_executable(test_app_library tests/test_app_library.c src/platform_persistence.c
      src/platform_web_input.c src/platform_export.c src/rich_layout.c vendor/editbox.c)
    target_include_directories(test_app_library PRIVATE vendor)
    target_link_libraries(test_app_library PRIVATE editor_core SDL3::SDL3 SDL3_ttf::SDL3_ttf)
    target_compile_definitions(test_app_library PRIVATE EDITOR_FONT_PATH="assets/font.otf")
    add_dependencies(test_app_library notelab)
    if(MSVC)
      target_compile_options(test_app_library PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_app_library PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME app_library COMMAND test_app_library "${CMAKE_CURRENT_BINARY_DIR}/app-library-notes" "${EDITOR_FONT_PATH}")
    set_tests_properties(app_library PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    add_executable(test_editbox tests/test_editbox.c vendor/editbox.c)
    target_include_directories(test_editbox PRIVATE vendor)
    target_link_libraries(test_editbox PRIVATE SDL3::SDL3 SDL3_ttf::SDL3_ttf)
    if(MSVC)
      target_compile_options(test_editbox PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_editbox PRIVATE -UNDEBUG -Wall -Wextra)
      if(EDITOR_SANITIZERS)
        target_compile_options(test_editbox PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test_editbox PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
    add_test(NAME text_input COMMAND test_editbox "${EDITOR_FONT_PATH}")
    add_executable(test_rich_layout tests/test_rich_layout.c src/rich_layout.c)
    target_link_libraries(test_rich_layout PRIVATE editor_core SDL3::SDL3 SDL3_ttf::SDL3_ttf)
    if(MSVC)
      target_compile_options(test_rich_layout PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_rich_layout PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME rich_layout COMMAND test_rich_layout "${EDITOR_FONT_PATH}" "${EDITOR_EMOJI_FONT_PATH}")
    add_test(NAME app_smoke COMMAND notelab --smoke-test --notes "${CMAKE_CURRENT_BINARY_DIR}/smoke-notes")
    set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
  endif()
  if(EMSCRIPTEN)
    set_target_properties(notelab PROPERTIES SUFFIX ".html")
    set_property(TARGET notelab APPEND PROPERTY LINK_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/platform/web/shell.html")
    target_link_options(notelab PRIVATE
      "-sALLOW_MEMORY_GROWTH=1" "-sFORCE_FILESYSTEM=1" "-lidbfs.js"
      "SHELL:--preload-file ${EDITOR_FONT_PATH}@/assets/font.otf"
      "SHELL:--preload-file ${EDITOR_EMOJI_FONT_PATH}@/assets/emoji.ttf"
      "SHELL:--preload-file ${EDITOR_ASSETS_DIR}/OFL.txt@/assets/OFL.txt"
      "SHELL:--preload-file ${EDITOR_EMOJI_LICENSE_PATH}@/assets/OFL-Emoji.txt"
      "SHELL:--preload-file ${EDITOR_THIRD_PARTY_PATH}@/assets/THIRD_PARTY.md"
      "SHELL:--preload-file ${EDITOR_NOTICES_DIR}@/assets/licenses"
      "SHELL:--shell-file ${CMAKE_CURRENT_SOURCE_DIR}/platform/web/shell.html")
    add_custom_command(TARGET notelab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_THIRD_PARTY_PATH}" "$<TARGET_FILE_DIR:notelab>/THIRD_PARTY.md"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${EDITOR_NOTICES_DIR}" "$<TARGET_FILE_DIR:notelab>/licenses")
  elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(notelab PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.notelab.editor"
      XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
    target_sources(notelab PRIVATE "${EDITOR_ASSETS_DIR}/font.otf" "${EDITOR_ASSETS_DIR}/OFL.txt"
      "${EDITOR_EMOJI_BUNDLED_PATH}" "${EDITOR_EMOJI_LICENSE_PATH}")
    set_source_files_properties("${EDITOR_ASSETS_DIR}/font.otf" "${EDITOR_ASSETS_DIR}/OFL.txt"
      "${EDITOR_EMOJI_BUNDLED_PATH}" "${EDITOR_EMOJI_LICENSE_PATH}"
      PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets")
  elseif(APPLE AND EDITOR_DISTRIBUTION)
    set_target_properties(notelab PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/macos/Info.plist.in"
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.notelab.editor"
      MACOSX_BUNDLE_BUNDLE_NAME "NoteLab"
      MACOSX_BUNDLE_BUNDLE_VERSION "2"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
    target_sources(notelab PRIVATE "${EDITOR_ASSETS_DIR}/font.otf" "${EDITOR_ASSETS_DIR}/OFL.txt"
      "${EDITOR_EMOJI_BUNDLED_PATH}" "${EDITOR_EMOJI_LICENSE_PATH}")
    set_source_files_properties("${EDITOR_ASSETS_DIR}/font.otf" "${EDITOR_ASSETS_DIR}/OFL.txt"
      "${EDITOR_EMOJI_BUNDLED_PATH}" "${EDITOR_EMOJI_LICENSE_PATH}"
      PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets")
  elseif(NOT ANDROID)
    add_custom_command(TARGET notelab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:notelab>/assets"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_ASSETS_DIR}/font.otf" "$<TARGET_FILE_DIR:notelab>/assets/font.otf"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_ASSETS_DIR}/OFL.txt" "$<TARGET_FILE_DIR:notelab>/assets/OFL.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_EMOJI_BUNDLED_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/emoji.ttf"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_EMOJI_LICENSE_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/OFL-Emoji.txt")
  endif()
  if(APPLE AND (CMAKE_SYSTEM_NAME STREQUAL "iOS" OR EDITOR_DISTRIBUTION))
    target_sources(notelab PRIVATE "${EDITOR_THIRD_PARTY_PATH}" ${EDITOR_NOTICE_FILES})
    set_source_files_properties("${EDITOR_THIRD_PARTY_PATH}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets")
    foreach(notice IN LISTS EDITOR_NOTICE_FILES)
      file(RELATIVE_PATH relative "${EDITOR_NOTICES_DIR}" "${notice}")
      get_filename_component(directory "${relative}" DIRECTORY)
      set_source_files_properties("${notice}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets/licenses/${directory}")
    endforeach()
  elseif(NOT ANDROID AND NOT EMSCRIPTEN)
    add_custom_command(TARGET notelab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_THIRD_PARTY_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/THIRD_PARTY.md"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${EDITOR_NOTICES_DIR}" "$<TARGET_FILE_DIR:notelab>/assets/licenses")
  endif()
endif()

if(EDITOR_DISTRIBUTION)
  include(cmake/Packaging.cmake)
endif()
