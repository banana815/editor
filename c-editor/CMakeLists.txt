      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_EMOJI_BUNDLED_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/emoji.ttf"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_EMOJI_LICENSE_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/OFL-Emoji.txt")
  endif()
  if(APPLE AND (CMAKE_SYSTEM_NAME STREQUAL "iOS" OR EDITOR_DISTRIBUTION))
    target_sources(notelab PRIVATE "${EDITOR_THIRD_PARTY_PATH}" ${EDITOR_NOTICE_FILES})
    set_source_files_properties("${EDITOR_THIRD_PARTY_PATH}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets")
    foreach(notice IN LISTS EDITOR_NOTICE_FILES)
      file(RELATIVE_PATH relative "${EDITOR_NOTICES_DIR}" "${notice}")
      get_filename_component(directory "${relative}" DIRECTORY)
      set_source_files_properties("${notice}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets/licenses/${directory}")
    endforeach()
  elseif(NOT ANDROID AND NOT EMSCRIPTEN)
    add_custom_command(TARGET notelab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${EDITOR_THIRD_PARTY_PATH}" "$<TARGET_FILE_DIR:notelab>/assets/THIRD_PARTY.md"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${EDITOR_NOTICES_DIR}" "$<TARGET_FILE_DIR:notelab>/assets/licenses")
  endif()
endif()

  include(cmake/Packaging.cmake)
