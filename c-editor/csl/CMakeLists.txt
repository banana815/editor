cmake_minimum_required(VERSION 3.24)
project(NoteLabCSL LANGUAGES C)
if(MSVC)
  # Dependencies also contain UTF-8 source; do not interpret their comments or
  # literals with the host's legacy Windows code page in standalone builds.
  add_compile_options(/utf-8)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSL_SANITIZERS "Instrument the C citation processor" OFF)
include(../cmake/CSLDependencies.cmake)
if(NOT TARGET cjson)
  set(EDITOR_BUILD_APP OFF)
  include(../cmake/Dependencies.cmake)
endif()
include(CTest)
option(CSL_BUILD_TESTING "Build standalone citation processor tests" ${BUILD_TESTING})

set(CSL_MODULES xml output unicode style locale input number number_field format condition case date sort collation names position variable registry inline evaluate finalize disambiguation sort_evaluate page_field disambiguation_render cluster bibliography document)
set(CSL_SOURCES)
foreach(module IN LISTS CSL_MODULES)
  list(APPEND CSL_SOURCES "csl_${module}.c")
endforeach()
add_library(notelab_csl STATIC ${CSL_SOURCES})
set_target_properties(notelab_csl PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(notelab_csl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(notelab_csl PUBLIC expat::expat utf8proc::utf8proc cjson)
target_link_libraries(notelab_csl PRIVATE notelab_mojibake)
if(MSVC)
  target_compile_options(notelab_csl PRIVATE /W4 /utf-8)
else()
  target_compile_options(notelab_csl PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CSL_SANITIZERS)
    target_compile_options(notelab_csl PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(notelab_csl PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(CSL_BUILD_TESTING)
  set(CSL_TEST_MODULES ${CSL_MODULES} render_profiles json)
  foreach(module IN LISTS CSL_TEST_MODULES)
    add_executable(test_csl_${module} tests/test_csl_${module}.c)
    target_link_libraries(test_csl_${module} PRIVATE notelab_csl)
    if(EMSCRIPTEN)
      # Exercise documented worst-case buffers and recoverable allocation
      # failures instead of Emscripten's default fixed 16 MiB abort behavior.
      target_link_options(test_csl_${module} PRIVATE
        -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=268435456 -sABORTING_MALLOC=0)
      if(module STREQUAL "evaluate" OR module STREQUAL "finalize" OR module STREQUAL "render_profiles" OR module STREQUAL "sort_evaluate" OR module STREQUAL "disambiguation_render" OR module STREQUAL "cluster" OR module STREQUAL "bibliography" OR module STREQUAL "document")
        target_link_options(test_csl_${module} PRIVATE -sSTACK_SIZE=1048576)
      endif()
    endif()
    if(MSVC)
      target_compile_options(test_csl_${module} PRIVATE /UNDEBUG /utf-8)
    else()
      target_compile_options(test_csl_${module} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(NOT module STREQUAL "render_profiles")
      add_test(NAME csl_${module} COMMAND test_csl_${module})
    endif()
  endforeach()

  set(CSL_RESOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/resources")
  if(EMSCRIPTEN)
    foreach(module xml style locale number date names variable evaluate finalize render_profiles sort_evaluate disambiguation_render cluster bibliography document)
      target_link_options(test_csl_${module} PRIVATE
        "--embed-file" "${CSL_RESOURCE_ROOT}@/csl-resources")
    endforeach()
    set(CSL_RESOURCE_ROOT "/csl-resources")
  endif()
  set(CSL_STYLES
    "${CSL_RESOURCE_ROOT}/apa.csl"
    "${CSL_RESOURCE_ROOT}/ieee.csl"
    "${CSL_RESOURCE_ROOT}/china-national-standard-gb-t-7714-2015-numeric.csl")
  set(CSL_LOCALES
    "${CSL_RESOURCE_ROOT}/locales-en-US.xml"
    "${CSL_RESOURCE_ROOT}/locales-zh-CN.xml")
  add_test(NAME csl_real_xml COMMAND test_csl_xml ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_styles COMMAND test_csl_style ${CSL_STYLES})
  add_test(NAME csl_real_locales COMMAND test_csl_locale ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_numbers COMMAND test_csl_number ${CSL_LOCALES})
  add_test(NAME csl_real_dates COMMAND test_csl_date ${CSL_LOCALES})
  add_test(NAME csl_real_names COMMAND test_csl_names ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_variables COMMAND test_csl_variable ${CSL_STYLES})
  add_test(NAME csl_real_evaluation COMMAND test_csl_evaluate ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_render_profiles COMMAND test_csl_render_profiles ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_sort_keys COMMAND test_csl_sort_evaluate ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_disambiguation COMMAND test_csl_disambiguation_render "${CSL_RESOURCE_ROOT}/apa.csl" "${CSL_RESOURCE_ROOT}/locales-en-US.xml")
  add_test(NAME csl_real_clusters COMMAND test_csl_cluster ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_bibliography COMMAND test_csl_bibliography ${CSL_STYLES} ${CSL_LOCALES})
  add_test(NAME csl_real_document COMMAND test_csl_document ${CSL_STYLES} ${CSL_LOCALES})
endif()
